public final class MortarLab {

    static final Color BOARD = new Color(0x0d0e10);
    static final Color GRID = new Color(0x1c1e24);
    static final Color TEXT = new Color(0xd8dae0);
    static final List<List<LoadedShell>> STOCK = List.of(
            List.of(LoadedShell.NORMAL, LoadedShell.HEAVY),
            List.of(LoadedShell.NORMAL, LoadedShell.HEAVY, LoadedShell.NAPALM),
            List.of(LoadedShell.NORMAL, LoadedShell.HEAVY, LoadedShell.NAPALM, LoadedShell.FLARE),
            List.of(LoadedShell.NORMAL, LoadedShell.HEAVY, LoadedShell.NAPALM, LoadedShell.FLARE, LoadedShell.NUKE));

    public static void main(String[] args) throws Exception {
        File out = new File(args[0]);
        looks(new File(out, "mortar-looks.png"));
        closeup(new File(out, "mortar-closeup.png"));
        frames(new File(out, "mortar-frames.png"));
        fireGif(new File(out, "mortar-fire.gif"));
        arcs(new File(out, "mortar-arcs.png"));
        arcGif(new File(out, "mortar-arcs.gif"));
    }

    static Graphics2D pen(BufferedImage im) {
        Graphics2D g = im.createGraphics();
        g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g.setRenderingHint(RenderingHints.KEY_RENDERING, RenderingHints.VALUE_RENDER_QUALITY);
        g.setRenderingHint(RenderingHints.KEY_STROKE_CONTROL, RenderingHints.VALUE_STROKE_PURE);
        g.setColor(BOARD);
        g.fillRect(0, 0, im.getWidth(), im.getHeight());
        return g;
    }

    static void reset() {
        TowerFoundations.Lab.stretch = 1;
        TowerFoundations.Lab.crouch = 0;
        TowerFoundations.Lab.piston = 0;
        TowerFoundations.Lab.gauge = 0;
        TowerFoundations.Lab.ring = -1;
        TowerFoundations.Lab.dust = -1;
        TowerFoundations.Lab.steam = -1;
        TowerFoundations.Lab.flash = false;
        TowerFoundations.Lab.loaded = false;
        TowerFoundations.Lab.bulge = -1;
        TowerFoundations.Lab.pump = variant == 0;
    }

    static int variant = 0;

    static void cellGrid(Graphics2D g, double cx, double cy, double cell) {
        g.setColor(GRID);
        g.setStroke(new BasicStroke(1f));
        g.draw(new Rectangle2D.Double(cx - cell / 2, cy - cell / 2, cell, cell));
    }

    static void shadow(Graphics2D g, double cx, double cy, double cell, double r) {
        g.setColor(new Color(0, 0, 0, 110));
        g.fill(new Ellipse2D.Double(cx - r * cell, cy - r * cell * 0.9 + cell * 0.04, r * cell * 2, r * cell * 1.8));
    }

    static void paint(Graphics2D g, TowerFoundations.Foundation f, double cx, double cy, double cell) {
        Graphics2D h = (Graphics2D) g.create();
        h.translate(cx, cy);
        h.scale(cell, cell);
        TowerSprites.paintParts(h, f.parts(), cell);
        TowerSprites.paintParts(h, f.front(), cell);
        h.dispose();
    }

    static TowerPose pose(int tier) {
        List<LoadedShell> stock = STOCK.get(tier);
        return TowerPose.idle().withStock(stock).withLoad(List.of(LoadedShell.NORMAL)).reloading(1f);
    }

    static void label(Graphics2D g, String s, double x, double y, int size) {
        g.setFont(new Font(Font.SANS_SERIF, Font.PLAIN, size));
        g.setColor(TEXT);
        g.drawString(s, (float) x, (float) y);
    }

    static void looks(File file) throws Exception {
        int cell = 220;
        int pad = 30;
        int w = pad * 2 + cell * 4 + 40 * 3 + 110;
        int h = pad + (cell + 70) * 2 + 90;
        BufferedImage im = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g = pen(im);
        for (int row = 0; row < 2; row++) {
            label(g, row == 0 ? "today" : "proposed", 10, pad + row * (cell + 70) + cell / 2.0 + 10, 20);
            for (int t = 0; t < 4; t++) {
                double cx = 110 + pad + t * (cell + 40) + cell / 2.0;
                double cy = pad + 20 + row * (cell + 70) + cell * 0.62;
                cellGrid(g, cx, cy, cell);
                reset();
                TowerFoundations.Lab.loaded = true;
                TowerFoundations.Lab.gauge = 1;
                TowerFoundations.Lab.piston = 1;
                TowerFoundations.Lab.stretch = 0.9;
                TowerFoundations.Lab.crouch = 0.45;
                TowerFoundations.Foundation f = row == 0 ? TowerFoundations.labCurrent(t, pose(t))
                        : TowerFoundations.labFat(t, pose(t));
                paint(g, f, cx, cy, cell);
                if (row == 0) {
                    label(g, "Tier " + t, cx - 25, pad + 14, 18);
                }
            }
        }
        label(g, "board size, 36 px", 10, h - 30, 14);
        for (int row = 0; row < 2; row++) {
            for (int t = 0; t < 4; t++) {
                double cx = 200 + t * 50 + row * 260;
                double cy = h - 40;
                reset();
                TowerFoundations.Lab.loaded = true;
                TowerFoundations.Lab.stretch = 0.9;
                TowerFoundations.Lab.crouch = 0.45;
                paint(g, row == 0 ? TowerFoundations.labCurrent(t, pose(t)) : TowerFoundations.labFat(t, pose(t)),
                        cx, cy, 36);
            }
        }
        g.dispose();
        ImageIO.write(im, "png", file);
    }

    static void closeup(File file) throws Exception {
        int cell = 420;
        BufferedImage im = new BufferedImage(cell * 3 + 40, (int) (cell * 1.25), BufferedImage.TYPE_INT_ARGB);
        Graphics2D g = pen(im);
        int[] tiers = {0, 1, 3};
        for (int i = 0; i < 3; i++) {
            double cx = 20 + i * cell + cell / 2.0;
            double cy = cell * 0.75;
            cellGrid(g, cx, cy, cell);
            reset();
            TowerFoundations.Lab.loaded = true;
            TowerFoundations.Lab.gauge = 1;
            TowerFoundations.Lab.piston = 1;
            TowerFoundations.Lab.stretch = 0.94;
            TowerFoundations.Lab.crouch = 0.3;
            paint(g, TowerFoundations.labFat(tiers[i], pose(tiers[i])), cx, cy, cell);
        }
        g.dispose();
        ImageIO.write(im, "png", file);
    }

    /** One tick of the proposed cycle: {@code t} ticks since the shot, cooldown {@code cd}. */
    static String state(int t, int cd) {
        reset();
        int load = (int) (cd * 0.75);
        int pumps = cd - load - 1;
        if (t == 0) {
            TowerFoundations.Lab.flash = true;
            TowerFoundations.Lab.stretch = 1.24;
            TowerFoundations.Lab.crouch = -0.7;
            TowerFoundations.Lab.ring = 0;
            return "spit";
        }
        double[] stretch = {1.24, 0.80, 1.08, 0.96, 1.0};
        double[] crouch = {-0.7, 1.0, 0.25, 0.5, 0.15};
        if (t < 5) {
            TowerFoundations.Lab.stretch = stretch[t];
            TowerFoundations.Lab.crouch = crouch[t];
            TowerFoundations.Lab.dust = (t - 1) / 4.0;
        }
        if (t < 9) {
            TowerFoundations.Lab.ring = t / 9.0;
        }
        if (t >= load) {
            TowerFoundations.Lab.loaded = true;
        }
        if (t >= load) {
            int into = t - load;
            if (variant == 0) {
                int len = 8;
                double u = Math.min(1, into / (double) len);
                double e = u * u * (3 - 2 * u);
                TowerFoundations.Lab.piston = e;
                TowerFoundations.Lab.stretch = 1 - 0.12 * e;
                TowerFoundations.Lab.crouch = 0.5 * e;
                TowerFoundations.Lab.gauge = e;
                if (into >= len - 1 && into < len + 3) {
                    TowerFoundations.Lab.steam = (into - (len - 1)) / 4.0;
                }
                return into < len ? "pump" : "cocked";
            }
            TowerFoundations.Lab.pump = false;
            int len = 5;
            if (into < len) {
                double u = into / (double) (len - 1);
                TowerFoundations.Lab.bulge = u;
                TowerFoundations.Lab.stretch = 1 - 0.1 * u;
                TowerFoundations.Lab.crouch = 0.45 * u;
                TowerFoundations.Lab.gauge = u;
                return "gulp";
            }
            TowerFoundations.Lab.stretch = 0.9;
            TowerFoundations.Lab.crouch = 0.45;
            TowerFoundations.Lab.gauge = 1;
            return "cocked";
        }
        if (t == 1) {
            return "slam";
        }
        if (t < 5) {
            return "bounce";
        }
        if (t < load) {
            return t < load * 0.33 ? "smoke" : "carry";
        }
        return "loaded";
    }

    static void carried(Graphics2D g, int t, int cd, double cx, double cy, double cell) {
        int from = (int) (cd * 0.25);
        int to = (int) (cd * 0.75);
        if (t < from || t >= to) {
            return;
        }
        double u = (t - from) / (double) (to - from);
        double s = u * u * (3 - 2 * u);
        double ax = -TowerFoundations.MORTAR_RACK_GAP / 2;
        double ay = 0.2 - TowerFoundations.LIFT * 0.13;
        double[] m = TowerFoundations.Lab.mouth;
        double mx = m[0];
        double my = m[1] - TowerFoundations.LIFT * m[2];
        double x = ax + (mx - ax) * s;
        double y = ay + (my - ay) * s - 4 * 0.28 * s * (1 - s);
        TowerSprites.paintShell(g, LoadedShell.NORMAL, (float) (cx + x * cell), (float) (cy + y * cell),
                (float) (TowerFoundations.MORTAR_SHELL * cell));
    }

    static void frames(File file) throws Exception {
        int cell = 170;
        int cd = 69;
        int[][] ticks = {{50, 52, 54, 56, 59, 70, 0}, {1, 2, 3, 6, 15, 30, 45}};
        String[] rows = {"load, pump, wait, spit", "slam, bounce, smoke, reload"};
        int cols = 7;
        int w = cols * (cell + 20) + 20;
        int h = 2 * (cell + 160) + 20;
        BufferedImage im = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g = pen(im);
        for (int row = 0; row < 2; row++) {
            variant = 0;
            label(g, rows[row], 20, 20 + row * (cell + 160) + 18, 18);
            for (int i = 0; i < cols; i++) {
                int t = ticks[row][i];
                double cx = 20 + i * (cell + 20) + cell / 2.0;
                double cy = 20 + row * (cell + 160) + cell * 0.72 + 110;
                String name = t >= cd ? "waiting" : state(Math.min(t, cd - 1), cd);
                if (t >= cd) {
                    TowerFoundations.Lab.steam = 0.3;
                }
                paint(g, TowerFoundations.labFat(1, pose(1)), cx, cy, cell);
                carried(g, t, cd, cx, cy, cell);
                if (t == 0) {
                    double[] m = TowerFoundations.Lab.mouth;
                    TowerSprites.paintShell(g, LoadedShell.NORMAL, (float) (cx + m[0] * cell),
                            (float) (cy + (m[1] - TowerFoundations.LIFT * m[2]) * cell - 0.22 * cell),
                            (float) (5 / 32.0 * cell));
                }
                String when = t >= cd ? "ready" : t >= 50 ? (t < 51 ? "load-1" : "load+" + (t - 51)) : "shot+" + t;
                label(g, when + "  " + name, cx - cell / 2.0 + 4, 20 + row * (cell + 160) + 48, 14);
            }
        }
        variant = 0;
        g.dispose();
        ImageIO.write(im, "png", file);
    }

    static void fireGif(File file) throws Exception {
        int cell = 180;
        int cd = 69;
        List<BufferedImage> frames = new ArrayList<>();
        int wait = 12;
        for (int k = 0; k < cd + wait; k++) {
            BufferedImage im = new BufferedImage(cell * 2, (int) (cell * 2.3), BufferedImage.TYPE_INT_ARGB);
            Graphics2D g = pen(im);
            for (int v = 0; v < 1; v++) {
                variant = v;
                int t = k < wait ? cd - 1 : k - wait;
                double cx = cell + v * cell * 2;
                double cy = cell * 1.65;
                cellGrid(g, cx, cy, cell);
                String name = state(t, cd);
                if (k < wait) {
                    name = "waiting for a target";
                }
                paint(g, TowerFoundations.labFat(1, pose(1)), cx, cy, cell);
                carried(g, t, cd, cx, cy, cell);
                double[] m = TowerFoundations.Lab.mouth;
                if (k >= wait && t < 6) {
                    double lift = 0.22 + 1.1 * t;
                    TowerSprites.paintShell(g, LoadedShell.NORMAL, (float) (cx + m[0] * cell + t * 0.05 * cell),
                            (float) (cy + (m[1] - TowerFoundations.LIFT * m[2] - lift * 0.35) * cell),
                            (float) (5 / 32.0 * cell));
                }
                label(g, name, cx - cell + 12, 26, 18);
            }
            variant = 0;
            g.dispose();
            frames.add(im);
        }
        gif(file, frames, 50);
    }

    // ---- arcs -----------------------------------------------------------------------------------------------------

    /** Screen lift of a shell (cells) at share {@code s} of a flight {@code d} cells long, option {@code o}. */
    static double lift(int o, double d, double s, double launch) {
        return switch (o) {
            case 0 -> Math.min(2, 0.06 * d * d) * 4 * s * (1 - s);
            case 1 -> launch * (1 - s) + Math.min(2.6, 1.1 + 0.32 * d) * 4 * s * (1 - s);
            default -> launch * (1 - s) + Math.min(3.4, 1.7 + 0.42 * d) * 4 * s * (1 - s);
        };
    }

    static final String[] OPTIONS = {"A  today: one arc, from the ground", "B  steep: from the muzzle, peak 1.1 + 0.32 d",
        "C  towering: from the muzzle, peak 1.7 + 0.42 d"};

    static void arcs(File file) throws Exception {
        int cell = 44;
        int cols = 13;
        int rows = 9;
        int pw = cols * cell;
        int ph = rows * cell;
        BufferedImage im = new BufferedImage(pw * 3 + 80, ph + 70, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g = pen(im);
        for (int o = 0; o < 3; o++) {
            double ox = 20 + o * (pw + 20);
            double oy = 50;
            arcPanel(g, o, ox, oy, cell, cols, rows, -1);
            label(g, OPTIONS[o], ox, 30, 15);
        }
        g.dispose();
        ImageIO.write(im, "png", file);
    }

    static double[][] TARGETS = {{1.5, 0}, {3, 0}, {4.5, 0}, {6.5, 0}, {-2.5, 2.5}, {0, -3.5}, {-4, -1.5}};

    static void arcPanel(Graphics2D g, int o, double ox, double oy, int cell, int cols, int rows, double tick) {
        Graphics2D h = (Graphics2D) g.create();
        h.clip(new Rectangle2D.Double(ox, oy, cols * cell, rows * cell));
        h.setStroke(new BasicStroke(1f));
        for (int i = 0; i < cols; i++) {
            for (int j = 0; j < rows; j++) {
                h.setColor(GRID);
                h.draw(new Rectangle2D.Double(ox + i * cell, oy + j * cell, cell, cell));
            }
        }
        double tx = ox + 5.5 * cell;
        double ty = oy + 5.5 * cell;
        reset();
        TowerFoundations.Lab.loaded = tick < 0;
        TowerFoundations.Foundation f = TowerFoundations.labFat(1, pose(1));
        double[] m = TowerFoundations.Lab.mouth;
        double launch = -(m[1] - TowerFoundations.LIFT * m[2]);
        paint(h, f, tx, ty, cell);
        for (double[] t : TARGETS) {
            double d = Math.hypot(t[0], t[1]);
            Path2D path = new Path2D.Double();
            int n = 60;
            for (int i = 0; i <= n; i++) {
                double s = i / (double) n;
                double x = tx + t[0] * s * cell;
                double y = ty + t[1] * s * cell - lift(o, d, s, launch) * cell;
                if (i == 0) {
                    path.moveTo(x, y);
                } else {
                    path.lineTo(x, y);
                }
            }
            h.setColor(new Color(0xe8b04a));
            h.setStroke(new BasicStroke(1.4f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND, 1f, new float[] {3, 4},
                    0));
            if (tick < 0) {
                h.draw(path);
            }
            h.setColor(new Color(0xd04a3a));
            h.setStroke(new BasicStroke(1.5f));
            double ex = tx + t[0] * cell;
            double ey = ty + t[1] * cell;
            h.draw(new Line2D.Double(ex - 4, ey - 4, ex + 4, ey + 4));
            h.draw(new Line2D.Double(ex - 4, ey + 4, ex + 4, ey - 4));
            double ticks = d * 32 / 8;
            if (tick >= 0) {
                double s = (tick % (ticks + 6)) / ticks;
                if (s <= 1) {
                    double gx = tx + t[0] * s * cell;
                    double gy = ty + t[1] * s * cell;
                    double l = lift(o, d, s, launch);
                    h.setColor(new Color(0, 0, 0, 140));
                    double sr = 0.09 * cell * (1 - 0.4 * Math.min(1, l / 2.5));
                    h.fill(new Ellipse2D.Double(gx - sr, gy - sr * 0.6, sr * 2, sr * 1.2));
                    TowerSprites.paintShell(h, LoadedShell.NORMAL, (float) gx, (float) (gy - l * cell),
                            (float) (5 / 32.0 * cell));
                }
            }
        }
        h.dispose();
    }

    static void arcGif(File file) throws Exception {
        int cell = 40;
        int cols = 13;
        int rows = 9;
        int pw = cols * cell;
        int ph = rows * cell;
        List<BufferedImage> frames = new ArrayList<>();
        for (int k = 0; k < 34 * 2; k++) {
            BufferedImage im = new BufferedImage(pw * 3 + 80, ph + 60, BufferedImage.TYPE_INT_ARGB);
            Graphics2D g = pen(im);
            for (int o = 0; o < 3; o++) {
                double ox = 20 + o * (pw + 20);
                arcPanel(g, o, ox, 45, cell, cols, rows, k / 2.0);
                label(g, OPTIONS[o].substring(0, OPTIONS[o].indexOf(':')), ox, 28, 15);
            }
            g.dispose();
            frames.add(im);
        }
        gif(file, frames, 25);
    }

    static void gif(File file, List<BufferedImage> frames, int ms) throws Exception {
        ImageWriter writer = ImageIO.getImageWritersBySuffix("gif").next();
        try (ImageOutputStream os = ImageIO.createImageOutputStream(file)) {
            writer.setOutput(os);
            writer.prepareWriteSequence(null);
            for (BufferedImage frame : frames) {
                BufferedImage rgb = new BufferedImage(frame.getWidth(), frame.getHeight(), BufferedImage.TYPE_INT_RGB);
                rgb.getGraphics().drawImage(frame, 0, 0, null);
                ImageTypeSpecifier type = ImageTypeSpecifier.createFromRenderedImage(rgb);
                IIOMetadata meta = writer.getDefaultImageMetadata(type, null);
                String format = meta.getNativeMetadataFormatName();
                IIOMetadataNode root = (IIOMetadataNode) meta.getAsTree(format);
                IIOMetadataNode gce = new IIOMetadataNode("GraphicControlExtension");
                gce.setAttribute("disposalMethod", "none");
                gce.setAttribute("userInputFlag", "FALSE");
                gce.setAttribute("transparentColorFlag", "FALSE");
                gce.setAttribute("delayTime", Integer.toString(ms / 10));
                gce.setAttribute("transparentColorIndex", "0");
                root.appendChild(gce);
                IIOMetadataNode apps = new IIOMetadataNode("ApplicationExtensions");
                IIOMetadataNode app = new IIOMetadataNode("ApplicationExtension");
                app.setAttribute("applicationID", "NETSCAPE");
                app.setAttribute("authenticationCode", "2.0");
                app.setUserObject(new byte[] {1, 0, 0});
                apps.appendChild(app);
                root.appendChild(apps);
                meta.setFromTree(format, root);
                writer.writeToSequence(new IIOImage(rgb, null, meta), null);
            }
            writer.endWriteSequence();
        }
    }
