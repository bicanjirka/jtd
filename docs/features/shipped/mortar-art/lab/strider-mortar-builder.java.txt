
    // ==== LAB: fat strider mortar =====================================================================================

    static final class Lab {
        static double stretch = 1;
        static double crouch = 0;
        static double piston = 0;
        static double gauge = 0;
        static double ring = -1;
        static double dust = -1;
        static double steam = -1;
        static boolean flash = false;
        static boolean loaded = false;
        static boolean props = true;
        static double bulge = -1;
        static boolean pump = true;
        static double[] mouth = {0, 0, 0};
    }

    static Foundation labCurrent(int tier, TowerPose pose) {
        return Foundation.of(siegeMortar(tier, pose));
    }

    static Foundation labFat(int tier, TowerPose pose) {
        return Foundation.of(fatMortar(tier, pose));
    }

    static final double FAT_X = 0.0;
    static final double FAT_Y = -0.05;
    static final double FAT_LEAN = 0.22;
    static final double FAT_FOOT = 0.335;
    static final double[] FAT_FEET = {-130, -50, 30, 150};

    /** The height of the floor under a foot {@code rho} out from the centre: the deck, or the plinth step it lands on. */
    private static double floorAt(int tier, double rho) {
        double r = FOOTPRINT[tier];
        double deck = (r - tier * PLINTH_STEP) * 0.92;
        if (tier >= 1 && rho <= deck) {
            return mortarFloor(tier);
        }
        double z = 0;
        for (int i = 0; i < tier; i++) {
            if (r - i * PLINTH_STEP >= rho) {
                z = (i + 1) * STEP_HEIGHT;
            }
        }
        return z;
    }

    private static double fatY(double h) {
        return FAT_Y - FAT_LEAN * h;
    }

    private static Build fatMortar(int tier, TowerPose pose) {
        List<LoadedShell> stock = pose.stock().isEmpty() ? List.of(LoadedShell.NORMAL) : pose.stock();
        List<LoadedShell> load = pose.load().isEmpty() ? List.of(LoadedShell.NORMAL) : pose.load();
        Color mortar = colour(TowerLook.MORTAR);
        Tones iron = tones(0x5d6372);
        Tones dark = tones(0x363b47);
        Tones groove = tones(0x23262e);
        Tones bronze = Tones.of(mix(mortar, Color.WHITE, 0.12));
        Tones wood = tones(0x7a5a3a);
        Tones plank = Tones.of(mix(wood.shade(), new Color(0x2a2622), 0.4));
        Build b = plinth(tier, mortar, r -> ngon(0, 0, 8, r, 22.5));
        double r = b.bodyRadius;
        double z = b.top;
        if (tier == 0) {
            b.flat(ell(0, 0.0, 0.43, 0.40), 0, new Color(0x16130f));
            b.flat(ell(0.02, 0.01, 0.34, 0.31), 0, new Color(0x1e1a15));
            for (int i = 0; i < 4; i++) {
                double a = 22 + i * 90;
                b.flat(at(ell(0, 0, 0.05, 0.035), a, 0.30), 0, new Color(0x2a2620));
            }
        }
        if (tier >= 1) {
            double deck = r * 0.92;
            b.solid(ngon(0, 0, 8, deck, 22.5), z, MORTAR_DECK, plank);
            for (int i = -2; i <= 2; i++) {
                b.flat(box(i * deck * 0.36, 0, 0.006, deck * 1.7), z + MORTAR_DECK, plank.shade());
            }
            if (tier >= 2) {
                b.flat(ring(0, 0, deck, deck - 0.022), z + MORTAR_DECK, iron.body());
                for (int i = 0; i < 8; i++) {
                    b.flat(at(circ(0, 0, 0.012), 22.5 + i * 45, deck - 0.011), z + MORTAR_DECK, iron.highlight());
                }
            }
            z += MORTAR_DECK;
        }
        if (tier == 3) {
            int mark = b.mark();
            for (int i = 0; i < 5; i++) {
                double a = -180 + i * 45;
                b.solid(at(circ(0, 0, 0.018), a, r * 0.80), z, 0.14, iron);
                b.flat(at(circ(0, 0, 0.022), a, r * 0.80), z + 0.14, iron.highlight());
            }
            b.line(arcOver(r * 0.80, -180, 0, z + 0.12), 0, iron.shade(), 0.016);
            b.around(mark);
        }
        double c = Lab.crouch;
        double s = Lab.stretch;
        double fat = 1 / Math.sqrt(s);
        double drop = 0.03 * c;
        double hipLow = z + 0.075 - drop;
        double hipHigh = z + 0.145 - drop;
        // back legs, then the drum, the barrel, the front legs
        for (double a : FAT_FEET) {
            if (sin(a) < 0) {
                strideLeg(b, tier, a, z, hipLow, hipHigh, c, dark, bronze);
            }
        }
        if (Lab.dust >= 0) {
            dust(b, tier, z, true);
        }
        double hx = FAT_X;
        double hy = fatY(0.10);
        b.solid(ngon(hx, hy, 8, 0.155, 22.5), hipLow, hipHigh - hipLow, dark);
        b.flat(ring(hx, hy, 0.155, 0.132), hipHigh, bronze.body());

        double rise = hipHigh;
        double[] radii = {0.150, 0.160, 0.166, 0.168, 0.168, 0.168};
        int rings = 5 + (pose.has(TowerCue.LONG_BARREL) ? 1 : 0);
        double ringHigh = 0.088 * s;
        double gaugeZ = 0;
        double pumpBottom = 0;
        double pumpTop = 0;
        for (int k = 0; k < rings; k++) {
            double h0 = rise - z;
            double y = fatY(h0);
            double rad = radii[k] * (k == 0 ? 1 : fat);
            if (Lab.bulge >= 0) {
                double from = (rings - 1 - k) / (double) (rings - 1);
                rad *= 1 + 0.13 * Math.exp(-Math.pow((from - Lab.bulge) / 0.22, 2));
            }
            b.solid(circ(FAT_X, y, rad - 0.018), rise, 0.012 * s, groove);
            rise += 0.012 * s;
            y = fatY(rise - z);
            b.solid(circ(FAT_X, y, rad), rise, ringHigh * 0.78, k == 1 || k == rings - 1 ? bronze : iron);
            b.solid(circ(FAT_X, fatY(rise - z + ringHigh * 0.78), rad - 0.012), rise + ringHigh * 0.78,
                    ringHigh * 0.22, Tones.of(iron.highlight()));
            if (k == 1) {
                gaugeZ = rise + ringHigh * 0.45;
                pumpBottom = rise - 0.03;
            }
            if (k == 3) {
                pumpTop = rise + ringHigh * 0.3;

            }
            rise += ringHigh;
        }
        // side pump on the east
        double px = FAT_X + 0.172 * fat + 0.03;
        double py = fatY((pumpBottom + pumpTop) / 2 - z) + 0.02;
        if (Lab.pump) {
        b.solid(box(px - 0.03, py, 0.03, 0.03), pumpBottom + 0.03, 0.03, dark);
        b.solid(circ(px, py, 0.032), pumpBottom, pumpTop - pumpBottom, bronze);
        b.flat(ring(px, py, 0.032, 0.02), pumpTop, bronze.highlight());
        double rod = pumpTop + 0.02 + 0.10 * (1 - Lab.piston);
        b.line(open(px, py - LIFT * pumpTop, px, py - LIFT * rod), 0, iron.shade(), 0.016);
        b.line(open(px - 0.002, py - LIFT * pumpTop, px - 0.002, py - LIFT * rod), 0, iron.highlight(), 0.006);
        b.line(open(px - 0.045, py - LIFT * rod, px + 0.045, py - LIFT * rod), 0, dark.shade(), 0.022);
        b.line(open(px - 0.045, py - LIFT * rod - 0.005, px + 0.045, py - LIFT * rod - 0.005), 0, wood.highlight(),
                0.008);
        }
        if (Lab.steam >= 0) {
            double a = Lab.steam;
            b.glow(px + 0.03 + 0.04 * a, py - 0.01, 0.05 + 0.05 * a, pumpBottom + 0.05 + 0.12 * a,
                    alpha(new Color(0xdfe3ea), 0.75 * (1 - a)));
        }
        // gauge on the front face
        double gy = fatY(gaugeZ - z) + 0.150 * fat * 0.92 - LIFT * gaugeZ;
        double gx = FAT_X - 0.06;
        b.flat(circ(gx, gy, 0.034), 0, bronze.shade());
        b.flat(circ(gx, gy, 0.026), 0, new Color(0xe9e2cc));
        double needle = -210 + 240 * Lab.gauge;
        b.line(open(gx, gy, gx + cos(needle) * 0.022, gy + sin(needle) * 0.022), 0,
                Lab.gauge > 0.85 ? new Color(0xd0402a) : new Color(0x2a2620), 0.007);
        // muzzle
        double my = fatY(rise - z);
        double lipOuter = 0.182 * fat;
        double bore = 0.118;
        b.solid(circ(FAT_X, my, lipOuter), rise, 0.03, bronze);
        rise += 0.03;
        my = fatY(rise - z);
        Shape lip = ring(FAT_X, my, lipOuter, bore);
        Area farLip = new Area(lip);
        farLip.intersect(new Area(new Rectangle2D.Double(-2, -2, 4, 2 + my)));
        Area nearLip = new Area(lip);
        nearLip.subtract(farLip);
        b.solid(farLip, rise, 0.035, bronze);
        b.flat(circ(FAT_X, my, bore), rise + 0.035, new Color(0x0d0b09));
        b.flat(circ(FAT_X, my + 0.02, bore * 0.7), rise + 0.035, new Color(0x1a1512));
        if (Lab.loaded) {
            shell(b, load.getFirst(), FAT_X, my, MORTAR_SHELL * 1.15, rise + 0.035 + MORTAR_SHELL * 0.4);
        }
        b.solid(nearLip, rise, 0.035, bronze);
        b.flat(ring(FAT_X, my, bore + 0.018, bore), rise + 0.035, iron.shade());
        for (int i = 0; i < 8; i++) {
            double a = 22.5 + i * 45;
            if (sin(a) > 0) {
                b.flat(mv(at(circ(0, 0, 0.010), a, (lipOuter + bore + 0.018) / 2), FAT_X, my), rise + 0.035,
                        bronze.highlight());


            }
        }
        double top = rise + 0.035;
        Lab.mouth = new double[] {FAT_X, my, top};
        if (Lab.flash) {
            b.glow(FAT_X, my, 0.30, top + 0.06, alpha(new Color(0xffe0a0), 0.95));
            b.glow(FAT_X, my, 0.14, top + 0.06, alpha(Color.WHITE, 0.95));
        }
        if (Lab.ring >= 0 && Lab.ring < 0.45) {
            double a = Lab.ring / 0.45;
            for (int i = 0; i < 5; i++) {
                double ang = -150 + i * 30;
                double out = 0.10 + 0.16 * a;
                b.glow(FAT_X + cos(ang) * out, my + sin(ang) * out * 0.5, 0.07 + 0.06 * a, top + 0.02 + 0.10 * a,
                        alpha(new Color(0xb4b7c0), 0.7 * (1 - a)));
            }
        }
        if (Lab.ring >= 0) {

            double a = Lab.ring;
            double rz = top + 0.06 + 0.55 * Math.sqrt(a);
            double rr = 0.13 + 0.13 * a;
            Color smoke = alpha(new Color(0xc9ccd4), 0.8 * (1 - a));
            b.flat(ring(FAT_X, my, rr, rr * 0.55), rz, smoke);
            b.glow(FAT_X, my, rr * 1.2, rz, alpha(new Color(0xc9ccd4), 0.25 * (1 - a)));
        }
        for (double a : FAT_FEET) {
            if (sin(a) > 0) {
                strideLeg(b, tier, a, z, hipLow, hipHigh, c, dark, bronze);
            }
        }
        if (Lab.dust >= 0) {
            dust(b, tier, z, false);
        }
        // rack in front
        int count = stock.size();
        int row = Math.min(count, MORTAR_RACK_ROW);
        double rackY = 0.2;
        double rackWide = row * MORTAR_RACK_GAP + 0.03;
        double rz = floorAt(tier, 0.2);
        b.solid(box(0, rackY, rackWide, MORTAR_RACK_GAP + 0.02), rz, MORTAR_RACK_HEIGHT, wood);
        for (int sd = -1; sd <= 1; sd += 2) {
            b.solid(box(sd * (rackWide / 2 - 0.008), rackY, 0.016, MORTAR_RACK_GAP + 0.02),
                    rz + MORTAR_RACK_HEIGHT, 0.03, Tones.of(wood.shade()));
        }
        for (int i = 0; i < count; i++) {
            double[] p = mortarPiece(tier, i, count);
            rackedShell(b, stock.get(i), p[0], rackY + (p[1] - MORTAR_RACK_Y), MORTAR_SHELL, p[2] - mortarFloor(tier)
                    + rz);
        }
        if (Lab.props && tier == 0) {
            keg(b, -0.36, -0.03, 0, wood, iron);
        }
        b.top = top;
        return b;
    }

    private static void keg(Build b, double x, double y, double z, Tones wood, Tones iron) {
        b.solid(circ(x, y, 0.055), z, 0.10, wood);
        b.flat(ring(x, y + 0.0, 0.056, 0.05), z + 0.0, iron.shade());
        b.line(open(x - 0.055, y + 0.012 - LIFT * (z + 0.025), x + 0.055, y + 0.012 - LIFT * (z + 0.025)), 0,
                iron.shade(), 0.010);
        b.line(open(x - 0.055, y + 0.012 - LIFT * (z + 0.075), x + 0.055, y + 0.012 - LIFT * (z + 0.075)), 0,
                iron.shade(), 0.010);
        b.flat(circ(x, y, 0.05), z + 0.10, wood.highlight());
        b.flat(ring(x, y, 0.05, 0.042), z + 0.10, wood.shade());
        b.flat(circ(x + 0.012, y - 0.01, 0.012), z + 0.10, new Color(0x2a2018));
    }

    private static void strideLeg(Build b, int tier, double a, double z, double hipLow, double hipHigh, double c,
                                  Tones t, Tones joint) {
        double hy = fatY(0.10);
        double hipR = 0.13;
        double hipX = FAT_X + cos(a) * hipR;
        double hipY = hy + sin(a) * hipR * 0.95;
        double hipZ = hipHigh + 0.05 - 0.02 * c;
        hipR = 0.15;
        hipX = FAT_X + cos(a) * hipR;
        hipY = fatY(hipZ - z) + sin(a) * hipR * 0.95;
        double kneeR = 0.27 + 0.035 * c;
        double kneeX = FAT_X + cos(a) * kneeR;
        double kneeY = hy + sin(a) * kneeR;
        double kneeZ = hipZ + 0.10 - 0.07 * c;
        double footX = FAT_X + cos(a) * FAT_FOOT;
        double footY = hy + sin(a) * FAT_FOOT;
        double footZ = floorAt(tier, Math.hypot(footX, footY));
        double w = 0.062 + 0.004 * tier;
        b.line(open(hipX, hipY - LIFT * hipZ, kneeX, kneeY - LIFT * kneeZ), 0, t.shade(), w);
        b.line(open(hipX - w * 0.15, hipY - LIFT * hipZ - w * 0.22, kneeX - w * 0.15, kneeY - LIFT * kneeZ - w * 0.22),
                0, t.highlight(), w * 0.33);
        b.line(open(kneeX, kneeY - LIFT * kneeZ, footX, footY - LIFT * (footZ + 0.02)), 0, t.shade(), w * 0.8);
        b.line(open(kneeX - w * 0.12, kneeY - LIFT * kneeZ, footX - w * 0.12, footY - LIFT * (footZ + 0.02)), 0,
                t.highlight(), w * 0.26);
        b.flat(circ(kneeX, kneeY - LIFT * kneeZ, w * 0.66), 0, t.shade().darker());
        b.flat(circ(kneeX - w * 0.12, kneeY - LIFT * kneeZ - w * 0.12, w * 0.46), 0, t.body());
        b.flat(circ(kneeX - w * 0.2, kneeY - LIFT * kneeZ - w * 0.2, w * 0.16), 0, joint.highlight());
        b.solid(mv(rot(poly(-0.03, -0.035, 0.06, 0, -0.03, 0.035), a), footX, footY), footZ, 0.022, t);
    }

    private static void dust(Build b, int tier, double z, boolean back) {
        double a = Lab.dust;
        for (double f : FAT_FEET) {
            if ((sin(f) < 0) == back) {
                double x = FAT_X + cos(f) * FAT_FOOT;
                double y = fatY(0.10) + sin(f) * FAT_FOOT;
                double fz = floorAt(tier, Math.hypot(x, y));
                b.glow(x + cos(f) * 0.05 * a, y + sin(f) * 0.04 * a, 0.05 + 0.07 * a, fz + 0.02 + 0.04 * a,
                        alpha(new Color(0xa89a80), 0.6 * (1 - a)));
            }
        }
    }
